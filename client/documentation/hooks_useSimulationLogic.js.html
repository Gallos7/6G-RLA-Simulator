import { useState } from 'react';
import axios from 'axios';
import { PRESETS } from '../data/presets';

/**
 * Custom Hook: useSimulationLogic
 * Encapsulates the entire business logic and state management for the simulation frontend.
 *
 * Responsibilities:
 * 1. Manages application state (inputs, configuration, results, theme, history).
 * 2. Handles user interactions (form updates, scenario selection).
 * 3. Communicates with the Python Backend API (Run Simulation, Fetch History).
 * 4. Implements validation rules for experiment parameters.
 *
 * @returns {Object} An object containing all state variables and handler functions required by the UI.
 */
const useSimulationLogic = () => {
    // --- APP STATE ---
    const [selectedScenario, setSelectedScenario] = useState("custom");
    const [theme, setTheme] = useState("light");

    // --- SIMULATION EXECUTION STATE ---
    const [loading, setLoading] = useState(false); // Indicates if a simulation is currently running
    const [results, setResults] = useState(null);  // Stores the chart data returned by the backend

    // --- HISTORY STATE ---
    const [isHistoryOpen, setIsHistoryOpen] = useState(false); // Controls the visibility of the history drawer
    const [historyData, setHistoryData] = useState([]);        // Stores the list of past simulations

    // --- INPUT PARAMETERS STATE ---
    // Contains all the raw inputs required for the Erlang B calculation
    const [inputs, setInputs] = useState({
        K: 2,  // Number of Service Classes
        M: 5,  // Number of Network Links (Topology)
        load: 4.0, // Traffic Intensity (Erlang)
        a_resource: [80, 400, 70, 70, 70], // Capacity Vector (Channels per link)
        r_resource: [[5, 25, 6, 5, 4], [2, 15, 4, 3, 2]], // Resource Requirement Matrix (b_kj)
        step: 1.0, // Default step size
        enableNoise: false, // Flag to inject random noise into probability calculations
        noiseMantissa: 1,   // Noise scientific notation base
        noiseExponent: "06" // Noise scientific notation exponent (e.g., 1e-6)
    });

    // --- SIMULATION CONTROL CONFIGURATION ---
    // Defines the parameters for the experiment loop (X-Axis Variable, Target, Step)
    const [simConfig, setSimConfig] = useState({
        variable: "load", // The independent variable to vary (e.g., 'load', 'fronthaul')
        target: 10.0,     // The stop condition value
        step: 0.1         // The increment step per iteration
    });

    // --- HELPER: Smart Defaults ---
    // Automatically suggests reasonable step and target values based on the selected variable.
    // This improves UX by preventing invalid ranges (e.g., step 0.1 for integer capacity).
    const getSmartDefaults = (variable, currentInputs) => {
        let suggestedStep = 0.1;
        let suggestedTarget = 10.0;

        switch (variable) {
            case "load":
                suggestedStep = 0.5;
                suggestedTarget = parseFloat(currentInputs.load) + 3.0;
                break;
            case "fronthaul":
                suggestedStep = 10;
                suggestedTarget = (currentInputs.a_resource[0] || 80) + 40;
                break;
            case "server":
                suggestedStep = 50;
                suggestedTarget = (currentInputs.a_resource[1] || 400) + 200;
                break;
            case "access":
                suggestedStep = 10;
                suggestedTarget = (currentInputs.a_resource[2] || 70) + 50;
                break;
            default:
                suggestedStep = 0.1;
        }
        return { step: suggestedStep, target: suggestedTarget };
    };

    // --- EVENT HANDLERS ---

    // Updates the number of services (K) and dynamically resizes the Matrix (r_resource).
    const handleServiceCountChange = (val) => {
        let newK = parseInt(val);
        if (newK < 2) newK = 2; // Min limit
        if (newK > 6) newK = 6; // Max limit

        setInputs(prev => {
            let currentMatrix = [...prev.r_resource];
            if (newK > currentMatrix.length) {
                // Add new rows initialized with default '1' values if K increases
                while (currentMatrix.length < newK)
                    currentMatrix.push(new Array(prev.M).fill(1));
            } else if (newK < currentMatrix.length)
                // Truncate rows if K decreases
                currentMatrix = currentMatrix.slice(0, newK);
            return { ...prev, K: newK, r_resource: currentMatrix };
        });
    };

    // Updates a specific cell in the Resource Requirement Matrix.
    const handleMatrixChange = (row, col, val) => {
        const newVal = parseInt(val) || 0;
        setInputs(prev => {
            const newMatrix = prev.r_resource.map(r => [...r]);
            newMatrix[row][col] = newVal;
            return { ...prev, r_resource: newMatrix };
        });
    };

    // Loads a historical simulation state into the active view and closes the drawer.
    const handleLoadHistory = (item) => {
        if (loadStateFromHistory) {
            loadStateFromHistory(item);
            setIsHistoryOpen(false);
        }
    };

    // Updates the capacity vector based on dropdown selections.
    // Handles special case for 'antenna' where multiple links (2, 3, 4) update simultaneously.
    const handleResourceSelect = (category, value) => {
        const val = parseInt(value);
        setInputs(prev => {
            const newResources = [...prev.a_resource];
            if (category === "fronthaul") newResources[0] = val;
            else if (category === "server") newResources[1] = val;
            else if (category === "antenna") {
                newResources[2] = val;
                newResources[3] = val;
                newResources[4] = val;
            }
            return { ...prev, a_resource: newResources };
        });
    };

    // Loads a pre-defined scenario from presets.js
    const handleScenarioChange = (e) => {
        const scenarioKey = e.target.value;
        setSelectedScenario(scenarioKey);
        const selectedPreset = PRESETS[scenarioKey];
        if (selectedPreset) {
            setInputs(prev => ({
                ...prev,
                K: selectedPreset.K,
                M: selectedPreset.M,
                load: selectedPreset.load,
                a_resource: selectedPreset.a_resource,
                r_resource: selectedPreset.r_resource,
                enableNoise: selectedPreset.enableNoise,
                noiseMantissa: selectedPreset.noiseMantissa,
                noiseExponent: selectedPreset.noiseExponent
            }));
            setSimConfig({
                variable: selectedPreset.variable || "load",
                target: selectedPreset.target || 10.0,
                step: selectedPreset.step || 1.0
            });
        }
        setResults(null); // Clear previous results on scenario change
    };

    // Generic input handler for numeric fields.
    // Automatically switches scenario to 'custom' if user modifies manual inputs.
    const handleInputChange = (field, value) => {
        if (field === "noiseExponent") {
            setInputs(prev => ({ ...prev, [field]: value }));
            if (selectedScenario !== "custom") setSelectedScenario("custom");
            return;
        }
        if (value === "") {
            setInputs(prev => ({ ...prev, [field]: "" }));
            return;
        }
        const numVal = parseFloat(value);
        if (numVal < 1.0) return; // Basic validation
        setInputs(prev => ({ ...prev, [field]: value }));
        if (selectedScenario !== "custom") setSelectedScenario("custom");
    };

    const handleCheckboxChange = (category) => {
        setInputs(prev => ({ ...prev, [category]: !prev[category] }));
    };

    const toggleTheme = () => setTheme(prev => prev === "light" ? "dark" : "light");

    // Handles changes in the Experiment Setup section (Variable, Target, Step).
    const handleSimConfigChange = (field, value) => {
        if (field === "variable") {
            // Apply smart defaults when changing the independent variable
            const defaults = getSmartDefaults(value, inputs);
            setSimConfig({ variable: value, step: defaults.step, target: defaults.target });
            return;
        }
        if (value === "") {
            setSimConfig(prev => ({ ...prev, [field]: "" }));
            return;
        }
        const numVal = parseFloat(value);
        if (numVal < 0.1) return;
        setSimConfig(prev => ({ ...prev, [field]: value }));
    };

    // --- LOAD HISTORY FUNCTION ---
    // Restores the application state from a saved history object.
    const loadStateFromHistory = (historyItem) => {
        console.log("Loading history item:", historyItem);
        if (historyItem.inputs) setInputs(historyItem.inputs);

        // Handle config object naming compatibility (config vs simConfig in DB)
        if (historyItem.config) setSimConfig(historyItem.config);
        else if (historyItem.simConfig) setSimConfig(historyItem.simConfig);

        if (historyItem.results) setResults(historyItem.results);
        setSelectedScenario(historyItem.scenario || "custom");
    };

    // --- VALIDATION LOGIC ---
    let error = null;
    const stepNum = parseFloat(simConfig.step);
    const targetNum = parseFloat(simConfig.target);

    // Determine the starting value based on the currently selected variable
    let startValue = 0;
    if (simConfig.variable === "load") startValue = inputs.load;
    else if (simConfig.variable === "fronthaul") startValue = inputs.a_resource[0];
    else if (simConfig.variable === "server") startValue = inputs.a_resource[1];
    else if (simConfig.variable === "access") startValue = inputs.a_resource[2];

    const diff = targetNum - startValue;

    if (isNaN(stepNum) || stepNum <= 0) error = "Step must be a positive number.";
    else if (targetNum <= startValue) error = `Target must be greater than Start (${startValue}).`;
    else if (stepNum > diff) error = "Step is too large for this range.";

    // --- API CALLS ---

    // POST: /run_simulation
    // Sends the current state to the Python backend to execute the experiment.
    const runSimulation = async () => {
        if (error) {
            alert("Please fix the errors before running.");
            return;
        }
        setLoading(true);
        setResults(null);
        try {
            const payload = {
                inputs: inputs,
                config: simConfig,
                scenario: selectedScenario
            };
            const response = await axios.post('http://localhost:5000/run_simulation', payload);
            setResults(response.data);
        } catch (err) {
            console.error("Simulation Failed:", err);
            alert("Error connecting to Python Server. Is it running?");
        } finally {
            setLoading(false);
        }
    };

    // GET: /api/history
    // Retrieves past simulations from the backend database.
    const fetchHistory = async () => {
        try {
            const res = await fetch('http://localhost:5000/api/history');
            const data = await res.json();
            if (data.error) {
                alert("Error from server: " + data.error);
                return;
            }
            setHistoryData(data);
            setIsHistoryOpen(true);
        } catch (err) {
            console.error("Failed to load history", err);
            alert("Could not connect to History Server.");
        }
    };

    // Construct the final noise value (float) for potential display or debugging
    const finalNoiseValue = inputs.enableNoise
        ? parseFloat(`${inputs.noiseMantissa}e-${inputs.noiseExponent}`)
        : 0;

    return {
        inputs, selectedScenario, theme, simConfig, error, finalNoiseValue, loading, results, historyData, isHistoryOpen,
        runSimulation, toggleTheme, handleServiceCountChange, handleMatrixChange, handleResourceSelect, handleScenarioChange,
        handleInputChange, handleCheckboxChange, handleSimConfigChange, loadStateFromHistory, fetchHistory, setIsHistoryOpen, handleLoadHistory
    };
};

export default useSimulationLogic;
